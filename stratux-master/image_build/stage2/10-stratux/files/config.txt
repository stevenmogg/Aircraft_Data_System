# RPi /boot/firmware/config.txt

dtparam=audio=on
max_usb_current=1
usb_max_current_enable=1
dtparam=i2c_arm=on
dtparam=i2c1=on
dtparam=i2c1_baudrate=100000
dtparam=i2c_arm_baudrate=100000
dtparam=spi=on

# MPI3508 3.5" HDMI panel (480x320) on Pi 5.
# Enable real KMS so cmdline video= is honoured (firmware bcm2708_fb follows bad EDID).
dtoverlay=vc4-kms-v3d-pi5
max_framebuffers=2
disable_fw_kms_setup=1

# move RPi3 Bluetooth off of hardware UART to free up connection for GPS
#dtoverlay=disable-bt
enable_uart=1

# i2c serial support (only if the NXP SC16IS752 at 0x4d is actually populated).
# A missing chip makes the Pi 5 I2C controller time out (~1 s) and stalls the AHRS IMU.
#dtoverlay=sc16is752-i2c,int_pin=4,addr=0x4d,xtal=1843900

# Otherwise TTGO T-Motion will cause kernel panic on bookworm
dtoverlay=dwc2,dr_mode=host

# disable default (mmc0) behavior on the ACT LED.
dtparam=act_led_trigger=none
dtparam=act_led_activelow=off

# The below has been added as a proposed EMI reduction measure. Issue #573.
sdram_freq=450
core_freq=450
arm_freq=900

arm_64bit=1

# --- cozy pi5 fan cooling (begin) ---
# Onboard FAN header thermal trips (not Stratux GPIO fancontrol).
# Temps in millidegrees C; speed 0-255. Enclosed avionics curve.
dtparam=fan_temp0=40000
dtparam=fan_temp0_hyst=5000
dtparam=fan_temp0_speed=100
dtparam=fan_temp1=50000
dtparam=fan_temp1_hyst=5000
dtparam=fan_temp1_speed=150
dtparam=fan_temp2=60000
dtparam=fan_temp2_hyst=5000
dtparam=fan_temp2_speed=200
dtparam=fan_temp3=67500
dtparam=fan_temp3_hyst=5000
dtparam=fan_temp3_speed=255
# --- cozy pi5 fan cooling (end) ---

# --- cozy mcp2515 can0 (begin) ---
# MCP2515 on SPI0 CE0. INT GPIO25, 8 MHz crystal, SPI 500 kHz (this harness).
dtparam=spi=on
dtoverlay=mcp2515-can0,oscillator=8000000,interrupt=25,spimaxfrequency=500000
# --- cozy mcp2515 can0 (end) ---
